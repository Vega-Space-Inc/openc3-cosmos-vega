# Commands for the VEGA target are HTTP GET requests against the Elara-AI /
# Vega Space REST API (https://admin.vega.space/api/v1). Each command has no
# body - it just tells HttpClientInterface which path/method to hit and which
# TELEMETRY packet the JSON response should be decoded into.
#
# These are triggered automatically on an interval via the OPTION PERIODIC_CMD
# lines in plugin.txt, so you normally never send them manually. You can also
# send them by hand from Command Sender to force an immediate refresh.
#
# Auth uses a frontend API key (vgk_... prefix, plugin variable vega_api_token),
# which is scoped to ONE Vega user - not an org - so every one of these
# endpoints lives under /api/v1/frontend/* and is gated by that user's
# individually-approved organization access (FrontendOrganizationRequest).
# GET_WORKSPACE / GET_FORECASTING_SUMMARY are scoped to a single org, chosen
# via the vega_org_id plugin variable - see GET_APPROVED_ORGS to find valid ids.

COMMAND VEGA GET_HEALTH BIG_ENDIAN "Poll the Vega API health check endpoint (unauthenticated)"
  ACCESSOR HttpAccessor
  PARAMETER HTTP_PATH 0 0 DERIVED nil nil "/api/v1/health_check"
  PARAMETER HTTP_METHOD 0 0 DERIVED nil nil "GET"
  PARAMETER HTTP_PACKET 0 0 DERIVED nil nil "HEALTH"

  ACCESSOR HttpAccessor

  PARAMETER HTTP_METHOD 0 0 DERIVED nil nil "GET"

# On-demand only (not periodically polled) - driven by the ForecastTimeline
# widget when the user picks a satellite/ground station/date. Returns the
# actual interference_windows (start/end time, severity, per-band breakdown)
# for one satellite+ground_station+day, which is what the timeline chart is
# built from. SATELLITE_ID/GROUND_STATION_ID/DATE are real (non-DERIVED)
# parameters so the widget can supply different values on every send, e.g.:
#   cmd("VEGA GET_DAY_DETAIL with SATELLITE_ID 11324, GROUND_STATION_ID 2473, DATE '2026-08-30'")
COMMAND VEGA GET_DAY_DETAIL BIG_ENDIAN "Poll one satellite+ground_station+day's interference windows"
  ACCESSOR HttpAccessor
  PARAMETER HTTP_PATH 0 0 DERIVED nil nil "/api/v1/frontend/organizations/<%= vega_org_id %>/forecasting/day_detail"
  PARAMETER HTTP_METHOD 0 0 DERIVED nil nil "GET"
  PARAMETER HTTP_PACKET 0 0 DERIVED nil nil "DAY_DETAIL"
  PARAMETER HTTP_ERROR_PACKET 0 0 DERIVED nil nil "AUTH_ERROR"
  PARAMETER HTTP_HEADER_AUTHORIZATION 0 0 DERIVED nil nil "Bearer <%= vega_api_token %>"
  PARAMETER HTTP_HEADER_CONNECTION 0 0 DERIVED nil nil "close"
  APPEND_PARAMETER HTTP_QUERY_SATELLITE_ID 32 UINT MIN MAX 0 "Satellite id (from VEGA WORKSPACE SATELLITES_JSON)"
  APPEND_PARAMETER HTTP_QUERY_GROUND_STATION_ID 32 UINT MIN MAX 0 "Ground station id (from VEGA WORKSPACE GROUND_STATIONS_JSON)"
  APPEND_PARAMETER HTTP_QUERY_DATE 256 STRING "2026-01-01" "Date to query, YYYY-MM-DD"

# On-demand only - driven by the ForecastTimeline widget to fill the past
# portion of the chart. Returns measured historical interference (all bands
# combined) for one satellite+ground_station over an explicit time range,
# stitched server-side across live PostgreSQL and R2-archived coverage runs.
# Max range one request may cover: 7 days.
COMMAND VEGA GET_HISTORY BIG_ENDIAN "Poll historical interference for one satellite+ground_station over a time range"
  ACCESSOR HttpAccessor
  PARAMETER HTTP_PATH 0 0 DERIVED nil nil "/api/v1/frontend/organizations/<%= vega_org_id %>/forecasting/history"
  PARAMETER HTTP_METHOD 0 0 DERIVED nil nil "GET"
  PARAMETER HTTP_PACKET 0 0 DERIVED nil nil "HISTORY"
  PARAMETER HTTP_ERROR_PACKET 0 0 DERIVED nil nil "AUTH_ERROR"
  PARAMETER HTTP_HEADER_AUTHORIZATION 0 0 DERIVED nil nil "Bearer <%= vega_api_token %>"
  PARAMETER HTTP_HEADER_CONNECTION 0 0 DERIVED nil nil "close"
  APPEND_PARAMETER HTTP_QUERY_SATELLITE_ID 32 UINT MIN MAX 0 "Satellite id (from VEGA WORKSPACE SATELLITES_JSON)"
  APPEND_PARAMETER HTTP_QUERY_GROUND_STATION_ID 32 UINT MIN MAX 0 "Ground station id (from VEGA WORKSPACE GROUND_STATIONS_JSON)"
  APPEND_PARAMETER HTTP_QUERY_START_TIME 256 STRING "2026-01-01T00:00:00Z" "Range start, ISO8601 (inclusive)"
  APPEND_PARAMETER HTTP_QUERY_END_TIME 256 STRING "2026-01-02T00:00:00Z" "Range end, ISO8601 (exclusive; max 7 days after start)"
