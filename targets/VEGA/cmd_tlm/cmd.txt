# Commands for the VEGA target are HTTP GET requests against the Elara-AI /
# Vega Space REST API (https://admin.vega.space/api/v1). Each command has no
# body - it just tells HttpClientInterface which path/method to hit and which
# TELEMETRY packet the JSON response should be decoded into (HTTP_PACKET), and
# which packet a non-2xx response lands in (HTTP_ERROR_PACKET -> ERROR_RESPONSE).
#
# The polled commands are sent automatically: once at connect (OPTION
# CONNECT_CMD) and then on an interval (OPTION PERIODIC_CMD) - see plugin.txt.
# You can also send them by hand from Command Sender to force a refresh.
#
# AUTH: two sources for the Vega frontend API key (vgk_... prefix), neither of
# which is ever written to a COSMOS log or store:
#   1. Per request - the Timeline widget sends the user's own key in the
#      OBFUSCATEd HTTP_HEADER_AUTHORIZATION parameter (kept in that user's
#      browser, entered on the dashboard). This is the normal path.
#   2. Fallback - the COSMOS secret VEGA_API_KEY (Admin / Secrets), mounted into
#      the interface by `SECRET ENV VEGA_API_KEY VEGA_API_KEY`, used for the
#      background polls and any command sent without a token.
# targets/VEGA/lib/api_key_protocol.rb applies whichever is present on the wire
# and strips the token from the packet before the command logs are written.
# A missing/invalid key shows up as HTTP 401 in ERROR_RESPONSE.
#
# The key is scoped to ONE Vega user - not an org - so every one of these
# endpoints lives under /api/v1/frontend/* and is gated by that user's
# individually-approved organization access (FrontendOrganizationRequest).
# GET_WORKSPACE / GET_FORECASTING_SUMMARY are scoped to a single org, chosen
# via the vega_org_id plugin variable - see GET_APPROVED_ORGS to find valid ids.
#
# HTTP_PATH / HTTP_METHOD / HTTP_PACKET / HTTP_ERROR_PACKET are hidden from
# Command Sender by IGNORE_PARAMETER in target.txt.

# On-demand only (not periodically polled) - driven by the ForecastTimeline
# widget when the user picks a satellite/ground station/date. Returns the
# actual interference_windows (start/end time, severity, per-band breakdown)
# for one satellite+ground_station+day, which is what the timeline chart is
# built from. SATELLITE_ID/GROUND_STATION_ID/DATE are real (non-DERIVED)
# parameters so the widget can supply different values on every send, e.g.:
#   cmd("VEGA GET_DAY_DETAIL with SATELLITE_ID 11324, GROUND_STATION_ID 2473, DATE '2026-08-30'")
COMMAND <%= target_name %> GET_DAY_DETAIL BIG_ENDIAN "Poll one satellite+ground_station+day's interference windows"
  ACCESSOR HttpAccessor
  PARAMETER HTTP_PATH 0 0 DERIVED nil nil "/api/v1/frontend/organizations/<%= vega_org_id %>/forecasting/day_detail"
  PARAMETER HTTP_METHOD 0 0 DERIVED nil nil "GET"
  PARAMETER HTTP_PACKET 0 0 DERIVED nil nil "DAY_DETAIL"
  PARAMETER HTTP_ERROR_PACKET 0 0 DERIVED nil nil "ERROR_RESPONSE"
  # Optional per-request Bearer token - the Timeline widget sends the user's own
  # Vega frontend API key here ("Bearer vgk_..."). Empty = fall back to the
  # VEGA_API_KEY secret. OBFUSCATE masks it in Command Sender and the text log,
  # and api_key_protocol.rb strips it from the packet before the command logs
  # are written, so the key never lands in any COSMOS log or store.
  PARAMETER HTTP_HEADER_AUTHORIZATION 0 0 DERIVED nil nil "" "Per-request Bearer token (Vega frontend API key); empty = use the VEGA_API_KEY secret"
    OBFUSCATE
  APPEND_PARAMETER HTTP_QUERY_SATELLITE_ID 32 UINT MIN MAX 0 "Satellite id (from VEGA WORKSPACE SATELLITES_JSON)"
  APPEND_PARAMETER HTTP_QUERY_GROUND_STATION_ID 32 UINT MIN MAX 0 "Ground station id (from VEGA WORKSPACE GROUND_STATIONS_JSON)"
  APPEND_PARAMETER HTTP_QUERY_DATE 256 STRING "2026-01-01" "Date to query, YYYY-MM-DD"
