# Telemetry packets for the VEGA target decode JSON responses from the
# Elara-AI / Vega Space REST API (https://admin.vega.space/api/v1). Each
# packet uses ACCESSOR HttpAccessor JsonAccessor: HTTP_* items come from the
# response status/headers (via HttpAccessor), everything else is pulled out
# of the JSON body by JSONPath (via JsonAccessor, in each item's KEY).
#
# Response envelope from the API is {"data": {...}, "meta": {...}}, so most
# KEYs start with $.data. List endpoints return arrays - for a first pass
# this exposes a "first record" snapshot (index [0]) since COSMOS telemetry
# items are fixed-schema. See the plugin README for how to extend this
# (e.g. one packet per record, or a custom Vue tool) once you know which
# fields you actually want to watch.

TELEMETRY VEGA HEALTH BIG_ENDIAN "Vega API health check (no auth required)"
  ACCESSOR HttpAccessor JsonAccessor
  ITEM HTTP_STATUS 0 0 DERIVED "HTTP response status code"
    LIMITS DEFAULT 1 ENABLED 199 199 300 300
  APPEND_ITEM STATUS 256 STRING "Health status string, expect 'healthy'"
    KEY $.status
  APPEND_ITEM ENVIRONMENT 256 STRING "Deployment environment"
    KEY $.environment
  APPEND_ITEM VERSION 256 STRING "Deployed build/commit version"
    KEY $.version
  APPEND_ITEM UPTIME_SEC 64 FLOAT "API process uptime in seconds"
    KEY $.uptime

# Every non-2xx response from any VEGA command lands here (HTTP_ERROR_PACKET in
# cmd.txt) - not only auth failures. The body is deliberately NOT parsed as
# JSON: Vega returns an HTML error page for some responses, so it is exposed
# raw via BinaryAccessor as a variable-length string (BODY 0 STRING = the rest
# of the buffer). HTTP_STATUS tells you which kind of error it was.
TELEMETRY VEGA ERROR_RESPONSE BIG_ENDIAN "Non-2xx response from any Vega endpoint (bad/expired key, no approved org access, rate limit, not found, server error). Raw body, see HTTP_STATUS for the code."
  ACCESSOR HttpAccessor BinaryAccessor
  ITEM HTTP_STATUS 0 0 DERIVED "HTTP response status code, e.g. 401 = bad/expired VEGA_API_KEY secret, 403 = no approved access to vega_org_id, 404 = unknown org / no data in range, 429 = rate limited, 5xx = Vega server error"
    LIMITS DEFAULT 1 ENABLED 199 199 300 300
  APPEND_ITEM BODY 0 STRING "Raw response body (HTML or JSON, unparsed)"

TELEMETRY VEGA DEMO_KEY BIG_ENDIAN "Vega's public demo API key: read-only access to the Demo Org, served by Vega so it can be rotated without a plugin release. Nothing secret - it is what any visitor gets."

TELEMETRY VEGA FORECASTING_SUMMARY BIG_ENDIAN "Interference/coverage forecast summary for the requested org - forecast run status plus the first satellite's most recent day"
  ACCESSOR HttpAccessor JsonAccessor
  ITEM HTTP_STATUS 0 0 DERIVED "HTTP response status code, 403 means no approved access to this org"
    LIMITS DEFAULT 1 ENABLED 199 199 300 300
  APPEND_ITEM ORG_ID 32 UINT "Organization id echoed back - lets a caller confirm this response is for the org it asked about (e.g. via an HTTP_PATH override), since periodic polls to the default vega_org_id keep updating this same packet in the background"
    KEY $.data.organization.id
  APPEND_ITEM SATELLITES_JSON 0 ARRAY "Full satellites array (id, name, norad, bands, forecast_available, latest_run_id, last_forecast_at, days, etc) - used by the ForecastTimeline widget to sort/disable satellites in its picker by forecast availability"
    KEY $.data.satellites
  APPEND_ITEM RUN_STATUS 256 STRING "Forecast run status, e.g. completed"
    KEY $.data.forecast_run.status
  # STRING rather than BOOL: STATE can't be declared on a BOOL item (TRUE/FALSE
  # is not a valid state value) and an integer type would fail to read the JSON
  # boolean, so the JSON true/false is read as the string "true"/"false" and
  # coloured by STATE. Displays as OK / STALE.
  APPEND_ITEM RUN_STALE 40 STRING "Whether the forecast run is stale (JSON true/false)"
    KEY $.data.forecast_run.stale
    STATE OK false GREEN
    STATE STALE true YELLOW
  APPEND_ITEM FIRST_SAT_NAME 256 STRING "Name of the first satellite in the forecast"
    KEY $.data.satellites[0].name
  APPEND_ITEM FIRST_SAT_FORECAST_AVAILABLE 8 BOOL "Whether a forecast is available for the first satellite"
    KEY $.data.satellites[0].forecast_available
  APPEND_ITEM FIRST_SAT_DAY_MAX_INTENSITY 32 UINT "Max interference intensity on the first satellite's first forecast day"
    KEY $.data.satellites[0].days[0].max_intensity
  APPEND_ITEM FIRST_SAT_DAY_MAX_SEVERITY 256 STRING "Max interference severity on the first satellite's first forecast day, e.g. low/medium/high"
    KEY $.data.satellites[0].days[0].max_severity
    STATE LOW low GREEN
    STATE MEDIUM medium YELLOW
    STATE HIGH high RED
  APPEND_ITEM FIRST_SAT_DAY_EVENT_COUNT 32 UINT "Interference event count on the first satellite's first forecast day"
    KEY $.data.satellites[0].days[0].event_count

# On-demand response to GET_DAY_DETAIL - see cmd.txt. This is the real windowed
# interference data (actual start/end times, not day-level aggregates) that
# the ForecastTimeline widget renders as colored bar segments, matching the
# same data the Vega web app itself uses for this view.
TELEMETRY VEGA DAY_DETAIL BIG_ENDIAN "Interference windows for one satellite+ground_station+day"
  ACCESSOR HttpAccessor JsonAccessor
  ITEM HTTP_STATUS 0 0 DERIVED "HTTP response status code, 403 means no approved access to this org"
    LIMITS DEFAULT 1 ENABLED 199 199 300 300
  APPEND_ITEM SATELLITE_ID 32 UINT "Satellite id echoed back"
    KEY $.data.satellite.id
  APPEND_ITEM SATELLITE_NAME 256 STRING "Satellite name echoed back"
    KEY $.data.satellite.name
  APPEND_ITEM GROUND_STATION_ID 32 UINT "Ground station id echoed back"
    KEY $.data.ground_station.id
  APPEND_ITEM GROUND_STATION_NAME 256 STRING "Ground station name echoed back"
    KEY $.data.ground_station.name
  APPEND_ITEM DATE 256 STRING "Date (YYYY-MM-DD) echoed back"
    KEY $.data.date
  APPEND_ITEM BANDS_JSON 0 ARRAY "Frequency bands this satellite operates on"
    KEY $.data.bands
  APPEND_ITEM INTERFERENCE_WINDOWS_JSON 0 ARRAY "Array of {id, start_time, end_time, affected_minutes, event_count, max_intensity, max_severity, bands, band_summary} - one entry per contiguous interference window on this day (day-level aggregate, not per-band)"
    KEY $.data.interference_windows
  APPEND_ITEM TONE_WARNING_MIN 32 UINT "Per-band interference count at/above which a minute is toned 'warning'"
    KEY $.data.tone_thresholds.warning_min
  APPEND_ITEM TONE_HIGH_MIN 32 UINT "Per-band interference count at/above which a minute is toned 'high'"
    KEY $.data.tone_thresholds.high_min
  APPEND_ITEM MINUTES_JSON 0 ARRAY "Array of {timestamp, minute, covered, counts: {band: count|null, ...}} - one entry per minute this day. A count of null means the run never aggregated that band (no reading); 0 means analysed and clear. The ForecastTimeline widget derives per-band interference segments from this (not from INTERFERENCE_WINDOWS_JSON, which is a day-level aggregate across all bands)."
    KEY $.data.minutes
